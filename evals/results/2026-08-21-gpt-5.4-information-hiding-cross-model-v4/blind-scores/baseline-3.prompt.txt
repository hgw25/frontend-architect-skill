+function resolveSourcePath(source: string | URL, root: string): string {

+async function readAndValidateFlags<TSchema extends FeatureFlagSchema>(
+  options: ResolvedFeatureFlagsOptions<TSchema>,
+): Promise<LoadedFlags> {
+  let raw: string
+
+  try {
+    raw = await readFile(options.sourcePath, 'utf8')
+  } catch (error) {
+    const detail = error instanceof Error ? error.message : String(error)
+    throw createDiagnostic(
+      `Cannot read feature flag source "${options.sourcePath}".`,
+      `Create the file or fix the path. Underlying error: ${detail}`,

+  }

+    sourceMapContent: JSON.stringify(publicFlags, null, 2),

+function parseFlagSource(raw: string, sourcePath: string): Record<string, unknown> {
+  let parsed: unknown
+
+  try {
+    parsed = JSON.parse(raw)
+  } catch (error) {
+    const detail = error instanceof Error ? error.message : String(error)
+    throw createDiagnostic(
+      `Cannot parse JSON from "${sourcePath}".`,
+      `Fix the JSON syntax. Underlying error: ${detail}`,
+    )
+  }

+  if (!isPlainObject(parsed)) {

+      'The JSON file must contain a top-level object such as { "checkoutV2": true }.',
+    )

+function projectPublicFlags<TSchema extends FeatureFlagSchema>(
+  source: Record<string, unknown>,
+  options: ResolvedFeatureFlagsOptions<TSchema>,
+): Record<string, FeatureFlagPrimitive> {
+  const publicFlags: Record<string, FeatureFlagPrimitive> = {}
+
+  for (const flagName of options.expose) {
+    const definition = options.schema[flagName]
+    if (!definition) {
+      throw createDiagnostic(
+        `Unknown exposed flag "${flagName}".`,
+        `Add "${flagName}" to the schema or remove it from expose: [...].`,
+      )
+    }
+
+    if (!(flagName in source)) {
+      throw createDiagnostic(
+        `Missing exposed flag "${flagName}" in "${options.sourcePath}".`,
+        `Add "${flagName}" to the JSON file or remove it from expose: [...].`,
+      )
+    }
+
+    const value = source[flagName]
+    assertFlagValue(flagName, value, definition, options.sourcePath)
+    publicFlags[flagName] = value
+  }
+
+  return publicFlags
+}

+function assertFlagValue(
+  flagName: string,
+  value: unknown,
+  definition: FeatureFlagDefinition,
+  sourcePath: string,
+): asserts value is FeatureFlagPrimitive {
+  if (definition.kind === 'boolean') {
+    if (typeof value !== 'boolean') {
+      throw createDiagnostic(
+        `Flag "${flagName}" in "${sourcePath}" must be a boolean.`,
+        `Received ${describeValue(value)}. Update the JSON value or change the schema.`,
+      )
+    }
+
+    return
+  }
+
+  if (definition.kind === 'number') {
+    if (typeof value !== 'number' || Number.isNaN(value)) {
+      throw createDiagnostic(
+        `Flag "${flagName}" in "${sourcePath}" must be a number.`,
+        `Received ${describeValue(value)}. Update the JSON value or change the schema.`,
+      )
+    }
+
+    return
+  }
+
+  if (definition.kind === 'string') {
+    if (typeof value !== 'string') {
+      throw createDiagnostic(
+        `Flag "${flagName}" in "${sourcePath}" must be a string.`,
+        `Received ${describeValue(value)}. Update the JSON value or change the schema.`,
+      )
+    }
+
+    return
+  }
+
+  if (typeof value !== 'string' || !definition.values.includes(value)) {
+    throw createDiagnostic(
+      `Flag "${flagName}" in "${sourcePath}" must be one of ${definition.values.map((item) => JSON.stringify(item)).join(', ')}.`,
+      `Received ${describeValue(value)}. Update the JSON value or widen the enum schema.`,
+    )
+  }
+}

+function findDuplicates(values: readonly string[]): string[] {
+  const seen = new Set<string>()
+  const duplicates = new Set<string>()
+
+  for (const value of values) {
+    if (seen.has(value)) {
+      duplicates.add(value)
+    }
+
+    seen.add(value)
+  }
+
+  return [...duplicates]
+}

+  if (value === null) {
+    return 'null'
+  }
+
+  return `${typeof value} ${String(value)}`

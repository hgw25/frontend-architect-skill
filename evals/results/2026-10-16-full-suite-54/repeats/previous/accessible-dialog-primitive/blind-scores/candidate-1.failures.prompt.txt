+# React Dialog primitive
+
+A headless modal Dialog for React 19. It owns interaction and accessibility policy while the consuming product owns all copy, actions, layout, and styling.
+
+## Public API
+
+Import either the compound namespace or individual components from the package root:
+
+```tsx
+import { Dialog } from '@example/react-dialog'
+
+<Dialog.Root defaultOpen={false} onOpenChange={(open, { reason }) => {}}>
+  <Dialog.Trigger className="trigger">Edit profile</Dialog.Trigger>
+  <Dialog.Portal>
+    <Dialog.Overlay className="overlay">
+      <Dialog.Content className="panel">
+        <Dialog.Title>Edit profile</Dialog.Title>
+        <Dialog.Description>Update your public information.</Dialog.Description>
+        <Dialog.Close>Cancel</Dialog.Close>
+      </Dialog.Content>
+    </Dialog.Overlay>
+  </Dialog.Portal>
+</Dialog.Root>
+```
+
+- `Root` accepts uncontrolled `defaultOpen` or controlled `open` plus `onOpenChange`. Change details report `trigger`, `close`, `escape`, or `outside`.
+- `Portal` renders into `document.body` by default and accepts a custom `container`.
+- `Trigger`, `Overlay`, `Content`, `Title`, `Description`, and `Close` forward refs and native DOM props, including `className`, `style`, and `data-*` styling hooks. The primitive ships no CSS.
+- `Content` accepts `initialFocusRef` and cancelable `onOpenAutoFocus`, `onCloseAutoFocus`, and `onEscapeKeyDown` hooks. A prevented consumer click also prevents the associated open/close action.
+- `Title` supplies the accessible name. `Description` supplies the accessible description. Alternatively, set `aria-label`, `aria-labelledby`, or `aria-describedby` directly on `Content`.
+
+Internal IDs are intentionally owned by the primitive so trigger, title, description, and content references cannot drift. Product labels and business actions are not part of this API.
+
+## Implementation boundaries
+
+- `Dialog.tsx` owns the React composition contract, controlled/uncontrolled state, portal lifecycle, ARIA wiring, and DOM event composition.
+- `dialog/focus.ts` owns focus candidate discovery, initial focus, and Tab wrapping.
+- `dialog/modalStack.ts` is the per-document coordinator. Only the top nested Dialog handles Escape or redirects focus; it also makes outside branches inert, restores their prior state, locks scrolling, and restores local trigger focus as layers close.
+- Browser globals are accessed only after mount. A server render emits the trigger and a stable `aria-expanded` state; open portal content mounts after hydration, avoiding a server/client tree mismatch.
+
+## Verification matrix
+
+| Contract | Automated evidence |
+| --- | --- |
+| Controlled opening and `onOpenChange` | The example's parent Dialog is controlled and exercised by Playwright |
+| Uncontrolled opening | The nested Dialog uses internal state and is exercised by Playwright |
+| Initial focus, Tab/Shift+Tab trap, programmatic focus escape | Chromium behavior test |
+| Escape closes only the top layer and restores each local trigger | Nested Chromium behavior test |
+| Role, accessible name/description, `aria-modal`, outside `inert` | Chromium accessibility assertions |
+| Server safety and `defaultOpen` hydration without mismatch | Node `renderToString` tests plus a production-build Chromium hydration test |
+| Public ESM entry and declarations | Library build plus a NodeNext consumer importing through package exports |
+| Consumer-owned appearance | Example styles only public class hooks; package build contains no CSS |
+
+Before publishing, add the browsers and React versions the package actually commits to supporting. The current release claim is React 19 and the configured Playwright Chromium runtime.
+
+## Migration
+
+Migrate one product Dialog at a time so rollback stays local:
+
+1. Keep the product component name and props. Replace its internals with the primitive parts, mapping its existing open state to `Root` and its existing DOM classes to `Trigger`, `Overlay`, and `Content`.
+2. Move the existing title, description, actions, analytics, and async business behavior into the product wrapper. Do not move those concerns into this package.
+3. Run that product's keyboard and browser tests. Verify initial focus, both Tab directions, Escape, outside interaction, trigger restoration, nested flows, and screen-reader name/description.
+4. Convert the next product after the wrapper is stable. Deprecate the old local implementation with a replacement example; remove it only when all callers are migrated.
+
+If a product must preserve a special Escape or focus rule, use the cancelable `Content` hooks in its wrapper and document that product policy there. Do not fork the focus stack.

+function isFocusable(element: HTMLElement) {
+  return (
+    !element.hidden &&
+    element.getAttribute('aria-hidden') !== 'true' &&
+    !element.closest('[inert]') &&
+    element.getClientRects().length > 0
+  )

+  return Array.from(container.querySelectorAll<HTMLElement>(FOCUSABLE_SELECTOR)).filter(

+  )

+export function focusFirst(container: HTMLElement, preferred?: HTMLElement | null) {
+  const target =
+    (preferred && container.contains(preferred) && isFocusable(preferred) ? preferred : null) ??
+    getFocusableElements(container)[0] ??
+    container
+
+  target.focus({ preventScroll: true })
+  return target

+export function keepTabFocusInside(event: KeyboardEvent, container: HTMLElement) {
+  if (event.key !== 'Tab') return

+  const focusable = getFocusableElements(container)
+  if (focusable.length === 0) {
+    event.preventDefault()
+    container.focus({ preventScroll: true })
+    return
+  }

+  const first = focusable[0]
+  const last = focusable[focusable.length - 1]

+    event.preventDefault()

+type ModalLayer = {
+  content: HTMLElement
+  handleKeyDown: (event: KeyboardEvent) => void
+  redirectFocus: () => void
+  restoreFocus: () => void
+}

+class ModalStack {
+  private readonly document: Document
+  private readonly layers: ModalLayer[] = []
+  private readonly changedElements = new Map<HTMLElement, ElementState>()
+  private originalBodyOverflow: string | null = null
+
+  constructor(document: Document) {
+    this.document = document
+  }
+
+  add(layer: ModalLayer) {
+    this.layers.push(layer)
+    if (this.layers.length === 1) {
+      this.document.addEventListener('keydown', this.handleKeyDown, true)
+      this.document.addEventListener('focusin', this.handleFocusIn, true)
+      this.originalBodyOverflow = this.document.body.style.overflow
+      this.document.body.style.overflow = 'hidden'
+    }
+    this.syncOutsideContent()
+
+    return () => {
+      const index = this.layers.indexOf(layer)
+      if (index === -1) return
+
+      this.layers.splice(index, 1)
+      this.syncOutsideContent()
+
+      if (this.layers.length === 0) {
+        this.document.removeEventListener('keydown', this.handleKeyDown, true)
+        this.document.removeEventListener('focusin', this.handleFocusIn, true)
+        this.document.body.style.overflow = this.originalBodyOverflow ?? ''
+        this.originalBodyOverflow = null
+      }
+
+      queueMicrotask(layer.restoreFocus)
+    }
+  }
+
+  isTop(content: HTMLElement | null) {
+    return content !== null && this.layers.at(-1)?.content === content
+  }
+
+  private readonly handleKeyDown = (event: KeyboardEvent) => {
+    this.layers.at(-1)?.handleKeyDown(event)
+  }
+
+  private readonly handleFocusIn = (event: FocusEvent) => {
+    const top = this.layers.at(-1)
+    if (!top || top.content.contains(event.target as Node)) return
+    top.redirectFocus()
+  }
+
+  private restoreChangedElements() {
+    for (const [element, state] of this.changedElements) {
+      element.inert = state.inert
+      if (state.ariaHidden === null) element.removeAttribute('aria-hidden')
+      else element.setAttribute('aria-hidden', state.ariaHidden)
+    }
+    this.changedElements.clear()
+  }
+
+  private syncOutsideContent() {
+    this.restoreChangedElements()
+
+    const top = this.layers.at(-1)
+    if (!top) return
+
+    let branch: HTMLElement | null = top.content
+    while (branch && branch !== this.document.body) {
+      const parent: HTMLElement | null = branch.parentElement
+      if (!parent) break
+
+      for (const sibling of Array.from(parent.children)) {
+        if (sibling === branch || !(sibling instanceof HTMLElement)) continue
+        this.changedElements.set(sibling, {
+          ariaHidden: sibling.getAttribute('aria-hidden'),
+          inert: sibling.inert,
+        })
+        sibling.inert = true
+        sibling.setAttribute('aria-hidden', 'true')
+      }
+
+      branch = parent
+    }
+  }

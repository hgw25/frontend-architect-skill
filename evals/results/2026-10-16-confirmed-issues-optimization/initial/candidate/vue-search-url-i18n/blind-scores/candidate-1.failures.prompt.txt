+export const supportedLocales = ['en-US', 'fr-FR'] as const
+
+export type SearchLocale = (typeof supportedLocales)[number]
+
+export type SearchState = {
+  query: string
+  page: number
+  locale: SearchLocale
+}
+
+const defaultSearchState: SearchState = {
+  query: '',
+  page: 1,
+  locale: 'en-US',
+}
+
+function isSearchLocale(value: string | null): value is SearchLocale {
+  return supportedLocales.some((locale) => locale === value)
+}
+
+function parsePage(value: string | null): number {
+  if (value === null || !/^\d+$/.test(value)) return defaultSearchState.page
+
+  const page = Number(value)
+  return Number.isSafeInteger(page) && page >= 1 ? page : defaultSearchState.page
+}
+
+export function readSearchState(url: URL): SearchState {
+  const locale = url.searchParams.get('locale')
+
+  return {
+    query: url.searchParams.get('q') ?? defaultSearchState.query,
+    page: parsePage(url.searchParams.get('page')),
+    locale: isSearchLocale(locale) ? locale : defaultSearchState.locale,
+  }
+}
+
+export function writeSearchState(url: URL, state: SearchState): URL {
+  const nextUrl = new URL(url)
+  nextUrl.searchParams.set('q', state.query)
+  nextUrl.searchParams.set('page', String(state.page))
+  nextUrl.searchParams.set('locale', state.locale)
+  return nextUrl

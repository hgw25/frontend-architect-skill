+# Headless Dialog primitive
+
+这个包把 Dialog 的状态、语义、焦点和模态生命周期作为共享契约，产品仍然拥有文案、按钮、布局与视觉样式。示例中的 “Edit profile”、表单和业务操作都只存在于 `App.tsx`，不属于 primitive。
+
+## 公开 API
+
+```tsx
+<Dialog defaultOpen={false} onOpenChange={(open, details) => {}}>
+  <DialogTrigger className="productTrigger">Edit profile</DialogTrigger>
+  <DialogPortal container={optionalContainer} className="layer">
+    <DialogOverlay className="productOverlay">
+      <DialogContent className="productDialog">
+        <DialogTitle>Edit profile</DialogTitle>
+        <DialogDescription>Update your public information.</DialogDescription>
+        {productContent}
+        <DialogClose>Cancel</DialogClose>
+      </DialogContent>
+    </DialogOverlay>
+  </DialogPortal>
+</Dialog>
+```
+
+- `Dialog` 是唯一的打开状态所有者。传 `open` 与 `onOpenChange` 时受控；否则使用 `defaultOpen`。变更回调同时提供 `trigger`、`close-button`、`escape-key` 或 `outside-pointer` 原因。
+- `DialogTrigger` 与 `DialogClose` 渲染原生 button，转发 ref 和原生属性，并在消费者 `preventDefault()` 后保留消费者决定。
+- `DialogPortal` 在 hydration 完成后挂到 `document.body` 或指定 `container`，服务端和客户端首帧都不访问浏览器全局。它的包装节点可接收 class、style 和 data 属性。
+- `DialogOverlay` 拥有指针外部关闭策略；`onPointerDownOutside` 可取消关闭。`DialogContent` 拥有 `dialog`/`alertdialog` 语义、焦点圈与 Escape；`onEscapeKeyDown` 可取消关闭。
+- `DialogTitle` 与 `DialogDescription` 产生稳定 ID 并连接 ARIA。无可见标题时必须给 Content 提供 `aria-label` 或 `aria-labelledby`；描述可用 `aria-describedby={undefined}` 明确省略。
+- `initialFocusRef`、`onOpenAutoFocus` 和 `onCloseAutoFocus` 是聚焦扩展点。阻止 auto-focus 事件后，消费者负责把焦点放到合适的位置。
+- 所有可见节点都接受原生 DOM 属性和 `className`，并暴露 `data-state`；包不发布 CSS，也不规定产品层级 token。
+
+唯一公开入口是包根目录，`exports` 同时指向 ESM 运行时与声明文件。React 和 React DOM 是 peer dependencies，以确保三个产品共享自己的 React 单例。`src/dialog/focus-management.ts` 是私有实现，不可从包路径导入。
+
+## 核心边界
+
+`Dialog.tsx` 负责组合 API、受控/非受控协议、稳定标注和 DOM 事件合成。`focus-management.ts` 按 Document 管理一个 Dialog 栈，仅栈顶响应 Escape 和 Tab；打开时隔离背景、锁定滚动，关闭时恢复原属性和焦点。嵌套 Dialog 关闭后只回到父 Dialog 内的本地触发器，父层仍然保持模态约束。
+
+Portal 只在 effect 后出现，因此 `defaultOpen` 的服务端输出包含展开状态的 trigger，但不输出依赖浏览器目标的 portal 内容；hydrate 首帧一致，随后在浏览器挂载。若产品需要首屏服务端输出 Dialog 内容，应不使用 Portal，并由应用保证该内容在 hydration 前不可交互。

+  dialogId: symbol
+  initialFocus?: HTMLElement | null
+  isolationRoot: HTMLElement

+type IsolationSnapshot = { ariaHidden: string | null; inert: boolean }
+type DocumentState = {
+  entries: FocusScopeEntry[]
+  isolation: Map<HTMLElement, IsolationSnapshot>
+  originalOverflow: string

+const tabbableSelector = [
+  'a[href]',
+  'area[href]',
+  'button:not([disabled])',
+  'input:not([disabled]):not([type="hidden"])',
+  'select:not([disabled])',
+  'textarea:not([disabled])',
+  'iframe',
+  'object',
+  'embed',
+  '[contenteditable]:not([contenteditable="false"])',
+  '[tabindex]:not([tabindex="-1"])',

+function isVisible(element: HTMLElement) {
+  const style = element.ownerDocument.defaultView?.getComputedStyle(element)
+  return element.getClientRects().length > 0 && style?.visibility !== 'hidden'
+}
+
+function getTabbableElements(container: HTMLElement) {
+  return Array.from(container.querySelectorAll<HTMLElement>(tabbableSelector))
+    .filter(
+      (element) =>
+        element.tabIndex >= 0 &&
+        !element.matches(':disabled') &&
+        !element.closest('[inert]') &&
+        isVisible(element),
+    )
+    .sort((left, right) => {
+      if (left.tabIndex === right.tabIndex) return 0
+      if (left.tabIndex === 0) return 1
+      if (right.tabIndex === 0) return -1
+      return left.tabIndex - right.tabIndex
+    })
+}

+  const target =
+    entry.initialFocus && entry.content.contains(entry.initialFocus) && isVisible(entry.initialFocus)
+      ? entry.initialFocus
+      : entry.content.querySelector<HTMLElement>('[autofocus]') ??
+        getTabbableElements(entry.content)[0] ??
+        entry.content

+function trapTab(entry: FocusScopeEntry, event: KeyboardEvent) {
+  const tabbable = getTabbableElements(entry.content)
+  if (tabbable.length === 0) {
+    event.preventDefault()
+    entry.content.focus({ preventScroll: true })
+    return

+  const last = tabbable[tabbable.length - 1]

+  if (event.shiftKey && (active === first || !entry.content.contains(active))) {
+    event.preventDefault()
+    last.focus({ preventScroll: true })

+  }

+import { readFile } from 'node:fs/promises'
+import path from 'node:path'
+
+import type { Page } from '@playwright/test'
+
+export async function loadBuiltApp(page: Page, entry: 'controlled' | 'hydration' | 'index' = 'index') {
+  await page.route('http://dialog.test/**', async (route) => {
+    const pathname = new URL(route.request().url()).pathname.replace(/^\//, '') || 'index.html'
+    const filePath = path.resolve('dist', pathname)
+    const contentType = pathname.endsWith('.html')
+      ? 'text/html'
+      : pathname.endsWith('.css')
+        ? 'text/css'
+        : 'text/javascript'
+    await route.fulfill({ body: await readFile(filePath), contentType })
+  })
+  await page.goto(`http://dialog.test/${entry === 'index' ? '' : `${entry}.html`}`)
+  const initialButton =
+    entry === 'controlled'
+      ? 'Open controlled dialog'
+      : entry === 'hydration'
+        ? 'Open hydrated dialog'
+        : 'Edit profile'
+  await page.getByRole('button', { name: initialButton }).waitFor()
+}

function groupBy(items, generateKey) {
  if (typeof generateKey === 'string') {
    const propertyName = generateKey
    generateKey = (item) => item[propertyName]
  }

  const result = {}
  let index = 0

  for (const item of items) {
    const key = generateKey(item, index++, items)
    if (result[key]) result[key].push(item)
    else result[key] = [item]
  }

  return result
}

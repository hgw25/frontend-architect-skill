+# Dialog Primitive
+
+## Core boundaries
+
+- Primitive remains headless: it owns state, focus, keyboard and aria wiring, but does not ship product copy, buttons or visual tokens.
+- `DialogPortal` mounts after hydration before portaling to `document.body`, which keeps SSR safe and avoids browser global access during server render.
+- `DialogContent` owns focus entry, tab loop, top-of-stack `Escape` handling and trigger restoration; nested dialogs are coordinated through an internal stack, not by consumers.
+- Consumers own styling through `className`, DOM props and composition. No theme contract is required.
+
+## Public API
+
+```tsx
+import {
+  DialogRoot,
+  DialogTrigger,
+  DialogPortal,
+  DialogOverlay,
+  DialogContent,
+  DialogTitle,
+  DialogDescription,
+  DialogClose,
+} from 'frontend-architect-dialog-browser-eval'
+```
+
+- `DialogRoot`: `open`, `defaultOpen`, `onOpenChange`, `modal`
+- `DialogTrigger`: default button trigger, or `asChild` for custom trigger elements
+- `DialogPortal`: optional `container`, `forceMount`
+- `DialogOverlay`: headless backdrop with outside-click close
+- `DialogContent`: dialog surface with aria, focus trap and `Escape` close
+- `DialogTitle` / `DialogDescription`: label and description bindings
+- `DialogClose`: close action, supports `asChild`
+
+## Package exports
+
+```json
+{
+  "exports": {
+    ".": "./src/index.ts"
+  }
+}
+```
+
+The stable public surface is `src/index.ts`; internal focus stack helpers stay private.
+
+## Test matrix
+
+- SSR closed render does not touch `document` or `window`
+- Uncontrolled open path preserves the existing fixture behavior
+- Initial focus lands on the first focusable element in content
+- `Tab` and `Shift+Tab` loop inside the active dialog
+- `Escape` closes only the top nested dialog, then restores local trigger focus
+- Parent dialog remains open after child dialog closes
+- Trigger focus is restored when the last active dialog closes
+- Overlay click closes the matching dialog without leaking into content clicks
+
+## Migration
+
+1. Replace product-local modal wrappers with `DialogRoot` + `DialogTrigger` + `DialogPortal`.
+2. Move backdrop and surface markup into `DialogOverlay` and `DialogContent`, keeping existing product styles on `className`.
+3. Replace ad hoc heading and copy wiring with `DialogTitle` and `DialogDescription`.
+4. Delete product-level `Escape`, outside-click and focus-trap code after parity verification.
+5. Roll out nested flows first in the products that currently show focus escape, then remove old shared modal utilities.

+const FOCUSABLE_SELECTOR = [
+  'a[href]',
+  'area[href]',
+  'button:not([disabled])',
+  'input:not([disabled]):not([type="hidden"])',
+  'select:not([disabled])',
+  'textarea:not([disabled])',
+  'iframe',
+  'object',
+  'embed',
+  '[contenteditable]',
+  '[tabindex]:not([tabindex="-1"])',
+].join(',')
+
+type DialogEntry = {
+  content: HTMLElement
+  token: symbol
+}
+
+const dialogStack: DialogEntry[] = []
+
+function isVisible(element: HTMLElement) {
+  const style = window.getComputedStyle(element)
+  return style.display !== 'none' && style.visibility !== 'hidden'
+}
+
+export function getFocusableElements(container: HTMLElement) {
+  return Array.from(container.querySelectorAll<HTMLElement>(FOCUSABLE_SELECTOR)).filter(
+    (element) => isVisible(element) && !element.hasAttribute('aria-hidden'),
+  )
+}
+
+export function focusFirstElement(container: HTMLElement) {
+  const preferred = container.querySelector<HTMLElement>('[data-autofocus]')
+  if (preferred && !preferred.hasAttribute('disabled')) {
+    preferred.focus()
+    return true
+  }
+
+  const elements = getFocusableElements(container)
+  if (elements.length > 0) {
+    elements[0].focus()
+    return true
+  }
+
+  if (!container.hasAttribute('tabindex')) {
+    container.setAttribute('tabindex', '-1')
+  }
+  container.focus()
+  return false
+}
+
+export function trapFocus(container: HTMLElement, event: KeyboardEvent) {
+  if (event.key !== 'Tab') {
+    return
+  }
+
+  const elements = getFocusableElements(container)
+  if (elements.length === 0) {
+    event.preventDefault()
+    container.focus()
+    return
+  }
+
+  const activeElement = document.activeElement instanceof HTMLElement ? document.activeElement : null
+  const currentIndex = activeElement ? elements.indexOf(activeElement) : -1
+  const lastIndex = elements.length - 1
+
+  if (event.shiftKey) {
+    if (currentIndex <= 0) {
+      event.preventDefault()
+      elements[lastIndex].focus()
+    }
+    return
+  }
+
+  if (currentIndex === lastIndex) {
+    event.preventDefault()
+    elements[0].focus()
+  }
+}
+
+export function pushDialogToStack(content: HTMLElement) {
+  const token = Symbol('dialog')
+  dialogStack.push({ content, token })
+  return token
+}
+
+export function removeDialogFromStack(token: symbol) {
+  const index = dialogStack.findIndex((entry) => entry.token === token)
+  if (index >= 0) {
+    dialogStack.splice(index, 1)
+  }
+}
+
+export function isTopDialog(token: symbol) {
+  return dialogStack.at(-1)?.token === token
+}
